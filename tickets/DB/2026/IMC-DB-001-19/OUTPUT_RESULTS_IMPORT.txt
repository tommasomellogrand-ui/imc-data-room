function hash(a){
 const s=a.map(clean).join("|");
 let h=2166136261;

 for(let i=0;i<s.length;i++){
  h^=s.charCodeAt(i);
  h=Math.imul(h,16777619);
 }

 const x=(h>>>0).toString(16).padStart(8,"0");
 return(x+x+x+x+x+x+x+x).slice(0,64);
}

/* GATEWAY */
async function gateway(body){
 const response=await fetch(GATEWAY_URL,{
  method:"POST",
  headers:{
   "Content-Type":"application/json",
   "Accept":"application/json",
   "X-IMC-Channel":"IMPORT"
  },
  body:JSON.stringify({
   ...body,
   channel:"IMPORT"
  })
 });

 const raw=await response.text();

 let json=null;
 try{
  json=raw?JSON.parse(raw):{};
 }catch(_){}

 if(!response.ok||!json||json.ok===false){
  const detail=
   (json&&(json.message||json.error))||
   raw||
   ("HTTP "+response.status);

  throw Error(
   "Gateway "+response.status+" · "+detail
  );
 }

 return json;
}

/* PRELOAD DUPLICATI */
async function preloadExisting(){
 setPhase("LETTURA DB");

 const dest=destination();
 let offset=0;
 const limit=5000;

 S.existingById.clear();
 S.existingIds.clear();
 S.existingFingerprints.clear();

 while(true){
  const json=await gateway({
   action:"read",
   game_world_id:S.gw,
   repository:RESULTS,
   target_database:dest.database,
   target_table:dest.table,
   route_mode:"explicit",
   limit,
   offset
  });

  if(!Array.isArray(json.rows))throw Error("READ senza rows: preload non affidabile");
  json.rows.forEach(indexExisting);

  if(!json.has_more)break;

  const next=Number(json.next_offset);

  if(!Number.isFinite(next)||next<=offset){
   throw Error("Paginazione READ non valida");
  }

  offset=next;
 }

 log(
  "DB caricato · "+
  S.existingIds.size+
  " fixture già presenti"
 );
}

function indexExisting(r){
 const id=Number(r.sm_fixture_id), fp=clean(r.fingerprint);
 if(!Number.isSafeInteger(id)||id<=0)throw Error("Fixture ID non valido nel preload/batch");
 if(!S.existingById.has(id))S.existingById.set(id,new Set());
 S.existingById.get(id).add(fp);
 S.existingIds.add(id);
 if(fp)S.existingFingerprints.add(fp);
}

function classifyExisting(r,byId=S.existingById,fingerprints=S.existingFingerprints){
 const id=Number(r.sm_fixture_id), fp=clean(r.fingerprint);
 if(!Number.isSafeInteger(id)||id<=0||!fp)throw Error("Identità/fingerprint riga non validi");
 if(byId.has(id)){
  const versions=byId.get(id);
  return versions.size===1&&versions.has(fp)?"IDENTICAL":"CHANGED_EXISTING";
 }
 return fingerprints.has(fp)?"DUPLICATE_CONTENT":"NEW";
}

function splitExisting(rows){
 const out={fresh:[],skipped:[],changed:[],duplicate:[]};
 const byId=new Map(S.existingById), fps=new Set(S.existingFingerprints);
 for(const r of rows){
  const status=classifyExisting(r,byId,fps);
  out[{NEW:"fresh",IDENTICAL:"skipped",CHANGED_EXISTING:"changed",DUPLICATE_CONTENT:"duplicate"}[status]].push(r);
  if(status==="NEW"){
   byId.set(Number(r.sm_fixture_id),new Set([clean(r.fingerprint)]));
   fps.add(clean(r.fingerprint));
  }
 }
 // Conflicting versions of the same NEW ID in one batch: withhold both.
 const conflicts=new Set(out.changed.map(r=>Number(r.sm_fixture_id)));
 out.fresh=out.fresh.filter(r=>{
  if(!conflicts.has(Number(r.sm_fixture_id)))return true;
  out.changed.push(r);return false;
 });
 return out;
}

async function insertRows(rows){
 if(!rows.length)return 0;

 const dest=destination();

 const json=await gateway({
  action:"insert_many",
  game_world_id:S.gw,
  repository:RESULTS,
  target_database:dest.database,
  target_table:dest.table,
  route_mode:"explicit",
  rows,
  atomic:true
 });

 const written=Number(json.written_rows);

 if(!Number.isFinite(written)){
  throw Error(
   "written_rows non dichiarato dal Gateway"
  );
 }

 if(written!==rows.length)throw Error("Scrittura parziale: interrompo senza retry automatico ("+written+"/"+rows.length+")");
 rows.forEach(indexExisting);

 S.saved+=written;

 return written;
}

/* RESULT PARSER */
const MONTH={
 gennaio:1,
 febbraio:2,
 marzo:3,
 aprile:4,
 maggio:5,
 giugno:6,
 luglio:7,
 agosto:8,
 settembre:9,
 ottobre:10,
 novembre:11,
 dicembre:12
};

function normalizeStageRound(value){
 return Object.prototype.hasOwnProperty.call(STAGE_DICTIONARY,value)?STAGE_DICTIONARY[value]:value;
}

function dateInfo(){
 const root=resultsRoot();

 const h=[
  ...liveAll(root,".scheduleScreen-Heading")
 ].find(visible);

 if(!h){
  return{
   match_date:null,
   competition_stage:null,
   competition_round:null
  };
 }

 const text=clean(h.textContent);

 const m=text.toLowerCase().match(
  /(?:lun|mar|mer|gio|ven|sab|dom)?\s*(\d{1,2})\s+(gennaio|febbraio|marzo|aprile|maggio|giugno|luglio|agosto|settembre|ottobre|novembre|dicembre)\s+(\d{4})/
 );

 if(!m){
  return{
   match_date:null,
   competition_stage:null,
   competition_round:null
  };
 }

 const date=
  m[3]+"-"+
  String(MONTH[m[2]]).padStart(2,"0")+"-"+
  String(Number(m[1])).padStart(2,"0");

 const prefix=clean(
  text.slice(
   0,
   text.toLowerCase().indexOf(m[0])
  ).replace(/[:·\-]\s*$/,"")
 );

 let stage=null;
 let round=null;

 if(prefix){
  const leg=prefix.match(
   /^(.*?)(?:\s+|[-·])\b(Andata|Ritorno)\b$/i
  );

  if(leg){
   stage=clean(leg[1])||null;
   round=clean(leg[2])||null;
  }else if(
   /finale|semifinale|quarti|ottavi|sedicesimi|qualificazion|finals|finali/i.test(prefix)
  ){
   stage=prefix;
  }else{
   round=prefix;
  }
 }

 return{
  match_date:date,
  competition_stage:normalizeStageRound(stage),
  competition_round:normalizeStageRound(round)
 };
}

function resultButtons(){
 const root=resultsRoot();

 const seen=new Set();

 return[

  const m=(

  ).match(
   /MATCHREPORT_[A-Za-z0-9_]*\s*\(\s*['"]?(\d+)/i

function decision(row,homeName,awayName){
 const textParts=[];
 let next=row.nextElementSibling;

 while(next){
  if(
   next.querySelector&&
   next.querySelector(
    '[id="matchResult"]'
   )
  ){
   break;
  }

  const t=clean(next.textContent);

  if(t){
   textParts.push(t);
  }

  next=next.nextElementSibling;
 }

 const text=clean(textParts.join(" "));

 const out={
  penalty_home_score:null,
  penalty_away_score:null,
  aggregate_home_score:null,
  aggregate_away_score:null
 };

 let m=text.match(
  /La squadra\s+(.+?)\s+vince ai rigori con il punteggio di\s+(\d+)\s*[-–]\s*(\d+)/i
 );

 if(m){
  const winner=clean(m[1]);
  const a=Number(m[2]);
  const b=Number(m[3]);

  if(norm(winner)===norm(homeName)){
   out.penalty_home_score=a;
   out.penalty_away_score=b;
  }else if(norm(winner)===norm(awayName)){
   out.penalty_home_score=b;
   out.penalty_away_score=a;
  }
 }

 m=text.match(
  /La squadra\s+(.+?)\s+vince con il punteggio aggregato di\s+(\d+)\s*[-–]\s*(\d+)/i
 );

 if(m){
  const winner=clean(m[1]);
  const a=Number(m[2]);
  const b=Number(m[3]);

  if(norm(winner)===norm(homeName)){
   out.aggregate_home_score=a;
   out.aggregate_away_score=b;
  }else if(norm(winner)===norm(awayName)){
   out.aggregate_home_score=b;
   out.aggregate_away_score=a;
  }
 }

 return out;
}

/* NAVIGATION */
function clickReal(e){
 if(!e)return;

 const token=
  "imc_click_"+
  Date.now()+"_"+
  Math.random().toString(36).slice(2);

 e.setAttribute(
  "data-imc-click-token",
  token
 );

 document.documentElement.removeAttribute(
  "data-imc-click-error"
 );

 const script=document.createElement("script");

 script.textContent=
  "(function(){"+
   "var el=document.querySelector('[data-imc-click-token=\\\""+
   token+
   "\\\"]');"+
   "if(!el)return;"+
   "try{"+
    "if(typeof el.onclick==='function'){"+
     "el.onclick.call(el);"+
    "}else{"+
     "var raw=el.getAttribute('onclick')||'';"+
     "if(raw){(new Function(raw)).call(el);}"+
     "else{el.click();}"+
    "}"+
   "}catch(e){"+
    "document.documentElement.setAttribute("+
     "'data-imc-click-error',"+
     "String(e&&e.message||e)"+
    ");"+
   "}"+
  "})();";

 (
  document.documentElement||
  document.head||
  document.body
 ).appendChild(script);

 script.remove();

 e.removeAttribute(
  "data-imc-click-token"
 );

 const err=
  document.documentElement.getAttribute(
   "data-imc-click-error"
  );

 document.documentElement.removeAttribute(
  "data-imc-click-error"
 );

 if(err){
  throw Error(err);
 }
}

async function waitForStableResultState(expected={},beforeIds=null,timeout=25000,openResults=false){
 const start=Date.now();let previous=null;
 S.clubIdentity=null;
 while(Date.now()-start<timeout){
  if(S.stop)throw Error("Importazione interrotta");
  if(openResults){
   const root=resultsRoot(),tab=screenRoot()?.querySelector("#resultsTableTab");
   if(root&&!visible(root)&&tab&&visible(tab))clickReal(tab);
  }
  let state=null;
  try{state=stateSnapshot()}catch(_){}
  const matches=state&&Object.entries(expected).every(([k,v])=>state.raw[k]===v)&&
   (beforeIds===null||state.ids!==beforeIds);
  if(matches&&previous===state.signature){S.clubIdentity=state.identity;return state;}
  previous=matches?state.signature:null;
  await sleep(500);
 }
 throw Error("Results non stabile/target non raggiunto: nessuna scrittura della pagina");
}

function previousSnapshot(){

async function goPrevious(ids){
 if(S.stop)throw Error("Importazione interrotta");
 if(!resultsRoot()||!visible(resultsRoot()))return {status:"REAL_ERROR",reason:"Results non visibili"};
 if(!prevButton())return {status:"END_OF_SECTION",reason:"Precedenti assente"};
 const before=previousSnapshot();
 if(!before)return {status:"REAL_ERROR",reason:"Results iniziali non verificabili"};
 const beforeIds=ids.slice().sort((a,b)=>a-b).join("|");
 if(before.ids!==beforeIds)return {status:"REAL_ERROR",reason:"Results cambiati prima del click"};
 try{clickReal(prevButton())}catch(e){return {status:"REAL_ERROR",reason:e.message||String(e)}}
 const start=Date.now();let last=null,stable=0,reason="DOM non verificabile";
 S.clubIdentity=null;
 while(Date.now()-start<25000){
  if(S.stop)throw Error("Importazione interrotta");
  let state=null;
  try{state=previousSnapshot()}catch(e){reason=e.message||String(e)}
  if(state){
   stable=state.signature===last?stable+1:1;
   last=state.signature;
   if(stable>=3){
    if(state.key!==before.key)return {status:"REAL_ERROR",reason:"Precedenti ha cambiato competizione/country/division canonica"};
    S.clubIdentity=state.identity;
    if(!state.ids)return {status:"END_OF_SECTION",reason:"Nessun Result giocato nel target"};
    if(state.ids===beforeIds){
     if(state.date!==before.date)return {status:"REAL_ERROR",reason:"Stesse fixture giocate con data diversa"};
     return {status:"END_OF_SECTION",reason:"Stesso turno e stesso fixture set"};
    }
    if(state.date>before.date)return {status:"REAL_ERROR",reason:"Precedenti ha raggiunto Results giocati successivi"};
    return {status:"MOVED_TO_PREVIOUS_RESULTS"};
   }
  }else{last=null;stable=0;}
  await sleep(500);
 }
 return {status:"REAL_ERROR",reason:"Precedenti: stato incoerente o non stabile · "+reason};
}

function plan(kind){
 const c=context();
 if(kind==="country"&&(worldStructure()==="SINGLE"||c.competition_group!=="DOMESTIC"))return null;
 if(kind==="division"&&c.sm_action!=="league")return null;
 const select=
  kind==="country"
   ?countrySelect()
   :divisionSelect();

 if(!select){
  return null;
 }

 const options=[
  ...select.options
 ].map((o,i)=>({
  value:String(o.value||""),
  text:clean(o.textContent),
  index:i,
  disabled:o.disabled
 }))
 .filter(o=>
  !o.disabled&&
  (
   kind==="division"
    ?/^division(?:e)?\s+\d+$/i.test(o.text)
    :o.value&&o.text
  )
 );

 if(!options.length){
  return null;
 }

 let pos=options.findIndex(
  o=>o.index===select.selectedIndex
 );

 if(pos<0){
  pos=0;
 }

 return options
  .slice(pos)
  .concat(
   options.slice(0,pos)
  );
}

async function switchTo(kind,target){
 const before=rawContext(),oldIds=pageIds().sort((a,b)=>a-b).join("|");
 const select=
  kind==="country"
   ?countrySelect()
   :divisionSelect();

 if(!select){
  throw Error(
   "Menu "+kind+" non trovato"
  );
 }

 const idx=[
  ...select.options
 ].findIndex(o=>
  String(o.value)===
  String(target.value)
 );

 if(idx<0)throw Error("Opzione target non disponibile: "+kind+" "+target.value);

 select.selectedIndex=idx;
 select.value=select.options[idx].value;

 if(typeof select.onchange==="function"){
  select.onchange.call(
   select,
   new Event(
    "change",
    {bubbles:true}
   )
  );
 }else{
  select.dispatchEvent(
   new Event(
    "change",
    {bubbles:true}
   )
  );
 }

 const expected={action_raw:before.action_raw};
 if(kind==="country")expected.country_raw=String(target.value);
 else {expected.division_raw=String(target.value);expected.country_raw=before.country_raw;}
 await waitForStableResultState(expected,oldIds,25000,true);

 return true;
}

async function processSection(){
 setPhase("SCANSIONE");

 let ids=await processPage();

 while(ids.length){
  setPhase("PRECEDENTI");

  const outcome=await goPrevious(ids);
  if(outcome.status==="END_OF_SECTION"){
   log("END_OF_SECTION · "+outcome.reason);
   break;
  }
  if(outcome.status!=="MOVED_TO_PREVIOUS_RESULTS"){
   throw Error("REAL_ERROR · "+outcome.reason);
  }
  ids=await processPage();
 }

 S.done++;
 updateUI();
}

async function run(){
 await waitForStableResultState();
 const countries=plan("country");

 if(countries){
  for(let i=0;i<countries.length;i++){
   if(S.stop){
    throw Error(
     "Importazione interrotta"
    );
   }

   if(i){
    setPhase("CAMBIO COUNTRY");

    await switchTo(
     "country",
     countries[i]
    );
   }

   const divisions=plan("division");

   if(divisions){
    for(let j=0;j<divisions.length;j++){
     if(S.stop){
      throw Error(
       "Importazione interrotta"
      );
     }

     if(j){
      setPhase("CAMBIO DIVISION");

      await switchTo(
       "division",
       divisions[j]
      );
     }

     await processSection();
    }
   }else{
    await processSection();
   }
  }

 }else{
  const divisions=plan("division");

  if(divisions){
   for(let i=0;i<divisions.length;i++){
    if(S.stop){
     throw Error(
      "Importazione interrotta"
     );
    }

    if(i){
     setPhase("CAMBIO DIVISION");

     await switchTo(
      "division",
      divisions[i]
     );
    }

    await processSection();
   }
  }else{
   await processSection();
  }
 }
}
